# H5 session helper (not in the repo): MACsec off / on for the latency arm without stopping macsecmgrd
# (orchagent Upholds= macsecmgrd, so systemctl stop restarts it at once). off: CONFIG_DB snapshot, then
# hdel PORT|<p> macsec on the four DAC ports (macsecmgrd removes SAs / SCs, engines to bypass). on: hset them back,
# wait for 8 open SAs, compare the whole CONFIG_DB with the snapshot.
set -u
. /root/sonic-test-lib.sh
S=/run/h5-macsec-off.tsv
case "$1" in
off)
  st_dump 4 > "$S"; echo "snapshot $(cut -f1 "$S" | sort -u | wc -l) keys"
  for p in $P4; do echo "PORT|$p macsec=$(redis-cli -n 4 hget "PORT|$p" macsec)"; redis-cli -n 4 hdel "PORT|$p" macsec > /dev/null; done
  for i in $(seq 30); do b=$(openbcm-diag extmacsec | grep -o 'bypass: [0-9]* of'); [ "$b" = "bypass: 64 of" ] && break; sleep 1; done
  echo "after $i s: $(openbcm-diag extmacsec | grep -E 'objects|engines' | tr '\n' ' ')" ;;
on)
  for p in $P4; do redis-cli -n 4 hset "PORT|$p" macsec "$(awk -F'\t' -v k="PORT|$p" '$1==k && $2=="macsec" {print $3}' "$S")" > /dev/null; done
  for i in $(seq 90); do [ "$(openbcm-diag extmacsec | grep -c ' open ')" = 8 ] && break; sleep 1; done
  echo "MKA: $(openbcm-diag extmacsec | grep -c ' open ') open SAs after $i s; $(openbcm-diag extmacsec | grep -E 'engines')"
  if diff <(st_dump 4 | grep -v '^CONFIG_DB_INITIALIZED') <(grep -v '^CONFIG_DB_INITIALIZED' "$S"); then echo "CONFIG_DB identical to the snapshot"; else echo "FAIL: CONFIG_DB differs"; exit 1; fi ;;
esac
