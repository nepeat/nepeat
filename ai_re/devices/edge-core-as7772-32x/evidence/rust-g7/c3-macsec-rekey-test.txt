#!/usr/bin/env bash
# H5 session helper (not in the repo): the VRF-pair topology of macsec-traffic-test.sh around another test
# (macsec-rekey-test.sh / macsec-wrongcak-test.sh need it since 3c84eb3's traffic test restores CONFIG_DB at its end).
# Snapshot + restore + whole-CONFIG_DB verification by sonic-test-lib.sh, exactly as the traffic tests.
#   bash h5-with-pairs.sh <script> [args...]
set -u
. /root/sonic-test-lib.sh
st_args
ST_PHASES=("${ST_PAIRS_PHASES[@]}")
export SONIC_CLI_IFACE_MODE=alias
mka() { for _ in $(seq "$1"); do [ "$(openbcm-diag extmacsec | grep -c ' open ')" = 8 ] && return 0; sleep 1; done; return 1; }
st_verify_extra() { if mka 60; then echo "== MKA: 8 SAs open after the restore"; else echo "== FAIL: MKA not up 60 s after the restore"; return 1; fi; }
st_pairs_check; st_refusals || exit 1
st_begin vrf-pairs arm
st_pairs_up || { echo "FAIL: setup did not complete"; exit 1; }
mka 60 || { echo "FAIL: MKA not up after the setup"; exit 1; }
ping -c 3 -W 2 10.0.1.1 >/dev/null; ping -c 3 -W 2 10.0.31.1 >/dev/null
echo "== wrapper: topology up, running: $*"
bash "$@"; rc=$?
echo "== wrapper: test exit $rc"
exit $rc
