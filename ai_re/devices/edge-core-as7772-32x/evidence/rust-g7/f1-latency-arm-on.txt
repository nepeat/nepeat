#!/usr/bin/env bash
# H5 session helper (not in the repo): X3's link-down latency cycles. No diag-socket use in here.
#   bash h5-lat.sh <n>     n cycles, alternating Ethernet4 (ce1 -> ce0) and Ethernet124 (ce31 -> ce30)
# CONFIG_DB PORT|<far> admin_status down, 1.5 s, up, then 2.5 s + random 0-0.6 s. Prints host times around each hset.
set -u
n=${1:-20}
for p in Ethernet4 Ethernet124; do echo "BEFORE PORT|$p admin_status=$(redis-cli -n 4 hget "PORT|$p" admin_status)"; done
for i in $(seq "$n"); do
  if [ $((i % 2)) = 1 ]; then far=Ethernet4; else far=Ethernet124; fi
  a=$(date +%s.%N); redis-cli -n 4 hset "PORT|$far" admin_status down > /dev/null; b=$(date +%s.%N)
  echo "CYCLE $i $far down $a $b"
  sleep 1.5
  a=$(date +%s.%N); redis-cli -n 4 hset "PORT|$far" admin_status up > /dev/null; b=$(date +%s.%N)
  echo "CYCLE $i $far up $a $b"
  j=$(awk -v r=$RANDOM 'BEGIN { printf "%.3f", 2.5 + (r % 601) / 1000 }')
  echo "JITTER $i $j"
  sleep "$j"
done
sleep 3
for p in Ethernet4 Ethernet124; do echo "AFTER PORT|$p admin_status=$(redis-cli -n 4 hget "PORT|$p" admin_status)"; done
