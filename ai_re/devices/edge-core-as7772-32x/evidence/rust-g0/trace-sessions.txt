# trace-run.sh (run on the switch)

#!/usr/bin/env bash
# agent H, G0 S4: trace one MACsec bring-up inside the running syncd: trace on, restart macsecmgrd
# (MACsec ports torn down, MKA again, MACsec objects made again), wait for 8 open gates, trace off, dump.
#   bash -l trace-run.sh <out-file-on-box>
set -u
out=$1
D() { openbcm-diag "$@"; }
echo "== $(date -Is) $(readlink /run/current-system)"
echo "== syncd pid $(systemctl show -P MainPID sonic-syncd), libsai: $(grep -o '/nix/store/[^ ]*libsai.so[^ ]*' /proc/$(systemctl show -P MainPID sonic-syncd)/maps | sort -u)"
D extmacsec | grep -E '^ce|bypass'
D "extmacsec trace clear" >/dev/null; D "extmacsec trace on"
t0=$(date +%s.%N)
systemctl restart sonic-macsecmgrd
for i in $(seq 90); do [ "$(D extmacsec | grep -c ' open ')" = 8 ] && break; sleep 1; done
echo "== 8 gates open after $i s; settling 15 s"
sleep 15
D "extmacsec trace off"
D "extmacsec trace dump" > "$out"
head -1 "$out"
D extmacsec | grep -E '^ce|bypass'
echo "== runs: $(grep -c '^# run' "$out"); lines $(wc -l < "$out"); masked writes: $(grep -c '\*\*\*\*' "$out"); failed runs: $(grep '^= ' "$out" | grep -vc '^= 0$')"
c=$(cat /var/lib/sonic-macsec/lab-test.cak); echo "== CAK string in dump: $(grep -ci -F "$c" "$out")"
