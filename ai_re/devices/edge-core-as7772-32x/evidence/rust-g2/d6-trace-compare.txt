# trace-compare.py
import sys, re, hashlib
tbl=sys.argv[1]
seqs=[]
for i,l in enumerate(open(tbl),1):
    if l.startswith('seq '): seqs.append((i,l.split()[1]))
def name(L):
    n=None
    for i,s in seqs:
        if i<=L: n=s
        else: break
    return n
def parse(f):
    runs=[];cur=None
    for l in open(f):
        l=l.rstrip('\n')
        m=re.match(r'# run (\d+): phy (0x[0-9a-f]+), (\d+) operations from line (\d+)',l)
        if m:
            cur={'phy':m.group(2),'name':name(int(m.group(4))),'ops':[]}; continue
        if cur is None: continue
        if l.startswith('= '):
            cur['st']=l[2:]; runs.append(cur); cur=None; continue
        cur['ops'].append(l)
    return runs
def periodic(n): return n.startswith('pn.') or n.startswith('stats-')
def summ(runs):
    ev={};per={};np=0;nops=0
    for r in runs:
        h=hashlib.sha1('\n'.join(r['ops']).encode()).hexdigest()[:12]
        if periodic(r['name']):
            per.setdefault((r['phy'],r['name']),set()).add(h); np+=1
        else:
            ev.setdefault(r['phy'],[]).append((r['name'],h,len(r['ops']),r['st'])); nops+=len([o for o in r['ops'] if o[:1] in 'RWD'])
    return ev,per,np,nops
a=parse(sys.argv[2]); b=parse(sys.argv[3])
ea,pa,npa,oa=summ(a); eb,pb,npb,ob=summ(b)
print(f"{sys.argv[2].split('/')[-1]} vs {sys.argv[3].split('/')[-1]}: event runs {sum(map(len,ea.values()))} / {sum(map(len,eb.values()))}, event bus operations {oa} / {ob}, periodic runs {npa} / {npb}")
d=0
for p in sorted(set(ea)|set(eb)):
    if ea.get(p)!=eb.get(p):
        d+=1; print('  phy',p,'differs'); 
        A=ea.get(p,[]);B=eb.get(p,[])
        for i in range(max(len(A),len(B))):
            x=A[i] if i<len(A) else None; y=B[i] if i<len(B) else None
            if x!=y: print('    ',i,x,'|',y)
pd=sum(1 for k in set(pa)|set(pb) if pa.get(k)!=pb.get(k))
print(f"  per-PHY event differences {d}, periodic variant differences {pd}; failed runs {sum(1 for r in a if r['st']!='0')} / {sum(1 for r in b if r['st']!='0')}")
if pd:
    for k in sorted(set(pa)|set(pb)):
        if pa.get(k)!=pb.get(k): print('   periodic',k,sorted(pa.get(k,[])),sorted(pb.get(k,[])))
seqlist=lambda e: [x[0] for x in e.get('0x0',[])]
print('  phy 0x0 event sequence (b):', ' '.join(seqlist(eb)))
